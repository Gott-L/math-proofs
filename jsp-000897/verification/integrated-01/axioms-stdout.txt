GottL897.dense_neighborhood {n r : ℕ} (hr : 4 ≤ r) (hn : 2 ≤ n) (G : SimpleGraph (Fin n)) [DecidableRel G.Adj]
  (hG : SimpleGraph.extremalNumber n ⊤ ≤ G.edgeFinset.card) :
  ∃ v,
    G.degree v = G.maxDegree ∧
      n ≤ 2 * G.degree v ∧
        SimpleGraph.extremalNumber (G.degree v) ⊤ ≤ (SimpleGraph.induce (G.neighborSet v) G).edgeFinset.card
